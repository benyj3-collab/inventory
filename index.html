/** * Google Apps Script - Inventory Count Backend * * הוראות התקנה: * 1. צור Google Sheet חדש * 2. לחץ על Extensions > Apps Script * 3. העתק את כל הקוד הזה לתוך עורך הסקריפט * 4. לחץ Deploy > New deployment * 5. בחר Type: Web app * 6. הגדר: Execute as: Me, Who has access: Anyone * 7. לחץ Deploy והעתק את כתובת ה-URL * 8. הדבק את הכתובת בהגדרות האפליקציה */ const SHEET_NAME = "ספירת מלאי"; const HEADERS = ["מזהה", "ברקוד", "מוצר", "עלות", "עלות עודכנה", "מחיר מכירה", "ספק", "כמות", "מכשיר", "תאריך"]; function doPost(e) { try { const data = JSON.parse(e.postData.contents); if (data.action === "add") { return addEntries(data); } return ContentService .createTextOutput(JSON.stringify({ success: false, error: "Unknown action" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ success: false, error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { // Allow GET for testing connectivity return ContentService .createTextOutput(JSON.stringify({ success: true, message: "Inventory API is running" })) .setMimeType(ContentService.MimeType.JSON); } function addEntries(data) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); // Create sheet with headers if it doesn't exist if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]); sheet.getRange(1, 1, 1, HEADERS.length) .setFontWeight("bold") .setBackground("#4a90d9") .setFontColor("#ffffff"); sheet.setFrozenRows(1); // Set RTL sheet.setRightToLeft(true); } const entries = data.entries || []; if (entries.length === 0) { return ContentService .createTextOutput(JSON.stringify({ success: true, added: 0 })) .setMimeType(ContentService.MimeType.JSON); } // Build rows const rows = entries.map(entry => [ entry.id || "", "'" + (entry.barcode || ""), // Force text format for barcode entry.name || "", entry.cost || 0, entry.costEdited ? "כן" : "לא", entry.price || 0, entry.supplier || "", entry.qty || 0, entry.device || data.device || "", entry.ts || new Date().toISOString() ]); // Append rows const lastRow = sheet.getLastRow(); sheet.getRange(lastRow + 1, 1, rows.length, HEADERS.length).setValues(rows); // Auto-resize columns for (let i = 1; i <= HEADERS.length; i++) { sheet.autoResizeColumn(i); } return ContentService .createTextOutput(JSON.stringify({ success: true, added: rows.length, total: sheet.getLastRow() - 1, device: data.device })) .setMimeType(ContentService.MimeType.JSON); } /** * Helper function to create the summary sheet * Run this manually from Apps Script to generate a summary */ function createSummary() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const dataSheet = ss.getSheetByName(SHEET_NAME); if (!dataSheet) return; // Get or create summary sheet let summarySheet = ss.getSheetByName("סיכום"); if (summarySheet) { summarySheet.clear(); } else { summarySheet = ss.insertSheet("סיכום"); } summarySheet.setRightToLeft(true); const summaryHeaders = ["ברקוד", "מוצר", "עלות", "מחיר מכירה", "ספק", "כמות כוללת", "מספר ספירות"]; summarySheet.getRange(1, 1, 1, summaryHeaders.length).setValues([summaryHeaders]); summarySheet.getRange(1, 1, 1, summaryHeaders.length) .setFontWeight("bold") .setBackground("#2ecc71") .setFontColor("#ffffff"); summarySheet.setFrozenRows(1); // Get all data const data = dataSheet.getDataRange().getValues(); const grouped = {}; for (let i = 1; i < data.length; i++) { const barcode = String(data[i][1]).replace(/^'/, ""); if (!grouped[barcode]) { grouped[barcode] = { name: data[i][2], cost: data[i][3], price: data[i][5], supplier: data[i][6], totalQty: 0, countNum: 0 }; } grouped[barcode].totalQty += Number(data[i][7]) || 0; grouped[barcode].countNum++; // Use latest cost if edited if (data[i][4] === "כן") { grouped[barcode].cost = data[i][3]; } } const rows = Object.entries(grouped).map(([barcode, g]) => [ "'" + barcode, g.name, g.cost, g.price, g.supplier, g.totalQty, g.countNum ]); if (rows.length > 0) { summarySheet.getRange(2, 1, rows.length, summaryHeaders.length).setValues(rows); } for (let i = 1; i <= summaryHeaders.length; i++) { summarySheet.autoResizeColumn(i); } SpreadsheetApp.getUi().alert(`סיכום נוצר: ${rows.length} מוצרים`); }
